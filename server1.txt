function getBearerToken(req: express.Request) {
  const header = req.headers.authorization || '';
  return header.startsWith('Bearer ') ? header.slice(7) : '';
}

const MUTATING_METHODS = new Set(['POST', 'PUT', 'PATCH', 'DELETE']);

async function recordAuditEvent(req: AuthRequest, status: number) {
  try {
    if (!MUTATING_METHODS.has(req.method)) return;
    const requestId = String(req.headers['x-ifc-request-id'] || req.headers['x-ifc-offline-mutation'] || '');
    await supabaseServer.from('audit_logs').insert({
      id: `audit-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`,
      action: req.method,
      method: req.method,
      route: req.path,
      status,
      user_id: req.user?.id || null,
      user_name: req.user?.username || 'المدير العام (Admin)',
      request_id: requestId || null,
      details: { source: req.headers['x-ifc-offline-mutation'] ? 'offline-sync' : 'online' },
    });
  } catch (e) {
    // Audit must never make a successful academy operation fail.
    console.warn('Audit log skipped:', e);
  }
}

async function recordSyncConflict(req: AuthRequest, error: any) {
  try {
    if (!(error instanceof SyncConflictError) && error?.status !== 409) return;
    const requestId = String(req.headers['x-ifc-request-id'] || req.headers['x-ifc-offline-mutation'] || '').trim();
    const body = req.body || {};
    const entityType = error.entityType || null;
    const entityId = error.entityId || null;
    const baseVersionRaw = body?._ifc_base_version;
    const localVersion = Number.isFinite(Number(baseVersionRaw)) ? Number(baseVersionRaw) : null;
    const serverVersion = Number.isFinite(Number(error.serverData?.version)) ? Number(error.serverData.version) : null;
    if (supabaseServer.from) {
      await supabaseServer.from('sync_conflicts').upsert({
        id: requestId || `conflict-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`,
        request_id: requestId || null,
        entity_type: entityType,
        entity_id: entityId,
        device_id: String(req.headers['x-ifc-device-id'] || req.headers['x-ifc-client-id'] || 'unknown'),
        local_version: localVersion,
        server_version: serverVersion,
        local_data: body,
        server_data: error.serverData || null,
        resolution: 'pending',
        created_at: new Date().toISOString(),
      }, { onConflict: 'id' });
    }
    if (requestId) {
      await supabaseServer.from('sync_operations').upsert({
        id: requestId, request_id: requestId, entity_type: entityType, entity_id: entityId,
        operation: req.method, device_id: String(req.headers['x-ifc-device-id'] || req.headers['x-ifc-client-id'] || 'unknown'),
        status: 'conflict', error: String(error.message || 'SYNC_CONFLICT'), payload: body,
        base_version: localVersion, result_version: serverVersion, created_at: new Date().toISOString(), completed_at: new Date().toISOString(),
      }, { onConflict: 'id' });
    }
  } catch (e) {
    console.warn('Sync conflict record skipped:', e);
  }
}

async function idempotencyMiddleware(req: AuthRequest, res: express.Response, next: express.NextFunction) {
  if (!MUTATING_METHODS.has(req.method)) return next();
  const requestId = String(req.headers['x-ifc-request-id'] || req.headers['x-ifc-offline-mutation'] || '').trim();
  if (!requestId) return next();

  try {
    // Atomic reservation: concurrent duplicate requests cannot both become the owner.
    const { data: inserted, error: insertError } = await supabaseServer.from('idempotency_keys').insert({
      request_id: requestId,
      route: req.path,
      method: req.method,
      status: 102,
      response: null,
      created_at: new Date().toISOString(),
      expires_at: new Date(Date.now() + 7 * 86400000).toISOString(),
    }).select('request_id').maybeSingle();

    if (insertError && insertError.code !== '23505') {
      // Older DBs without the table should not block normal writes.
      console.warn('Idempotency reservation skipped:', insertError.message);
    } else if (insertError?.code === '23505' || !inserted) {
      // Existing request: return the completed response, or wait briefly while the
      // original request finishes. Never execute the same mutation twice concurrently.
      for (let attempt = 0; attempt < 12; attempt++) {
        const { data: existing } = await supabaseServer.from('idempotency_keys')
          .select('route,method,status,response,expires_at').eq('request_id', requestId).maybeSingle();
        if (!existing) break;
        if (existing.expires_at && new Date(existing.expires_at).getTime() < Date.now()) {
          await supabaseServer.from('idempotency_keys').delete().eq('request_id', requestId);
          break;
        }
        if (existing.route && existing.method && (existing.route !== req.path || existing.method !== req.method)) {
          return res.status(409).json({ error: 'معرّف العملية مستخدم لطلب مختلف.', code: 'IDEMPOTENCY_KEY_REUSED', requestId });
        }
        if (existing?.response !== null && existing?.response !== undefined) {
          return res.status(Number(existing.status) || 200).json(existing.response);
        }
        if (existing && Number(existing.status) !== 102) {
          return res.status(Number(existing.status) || 200).json(existing.response ?? { success: true, duplicate: true, requestId });
        }
        await new Promise((resolve) => setTimeout(resolve, 250));
      }
      return res.status(409).json({ error: 'العملية نفسها قيد التنفيذ بالفعل. أعد المحاولة بعد لحظات.', code: 'IDEMPOTENCY_IN_PROGRESS', requestId });
    }
  } catch (error) {
    console.warn('Idempotency check unavailable; continuing request:', error);
  }

  const originalJson = res.json.bind(res);
  const originalSend = res.send.bind(res);
  let stored = false;
  const store = async (status: number, body: any) => {
    if (stored) return;
    stored = true;
    try {
      const serialized = JSON.stringify(body);
      const response = serialized.length <= 900_000 ? body : null;
      await supabaseServer.from('idempotency_keys').update({
        status, response, completed_at: new Date().toISOString(),
      }).eq('request_id', requestId);
    } catch {}
  };
  (res as any).json = (body: any) => { void store(res.statusCode || 200, body); return originalJson(body); };
  (res as any).send = (body: any) => {
    let parsed = body;
    if (typeof body === 'string') { try { parsed = JSON.parse(body); } catch {} }
    void store(res.statusCode || 200, parsed);
    return originalSend(body);
  };
  res.on('finish', () => { void recordAuditEvent(req, res.statusCode); });
  next();
}
async function ensureAdminCredentialsTable() {
  try {
    let existing = await getAdminCredentials();
    if (!existing) {
      const { data, error } = await supabaseServer.auth.admin.createUser({
        email: ADMIN_AUTH_EMAIL,
        password: DEFAULT_ADMIN_PASSWORD,
        email_confirm: true,
        user_metadata: { username: DEFAULT_ADMIN_USERNAME, role: 'admin' },
      });
      if (error || !data.user) throw error || new Error('Unable to create Supabase Auth admin');
      await saveAdminCredentials({ username: DEFAULT_ADMIN_USERNAME, authUserId: data.user.id, authEmail: ADMIN_AUTH_EMAIL });
      return;
    }

    if (existing.auth_user_id) {
      const { data } = await supabaseServer.auth.admin.getUserById(existing.auth_user_id);
      if (data.user) return;
    }

    const email = existing.auth_email || ADMIN_AUTH_EMAIL;
    const { data: users, error: listError } = await supabaseServer.auth.admin.listUsers({ page: 1, perPage: 100 });
    if (listError) throw listError;
    const existingAuthUser = users.users.find((u) => u.email?.toLowerCase() === email.toLowerCase());
    if (existingAuthUser) {
      await saveAdminCredentials({ username: existing.username || DEFAULT_ADMIN_USERNAME, authUserId: existingAuthUser.id, authEmail: email });
      return;
    }

    const { data, error } = await supabaseServer.auth.admin.createUser({
      email,
      password: DEFAULT_ADMIN_PASSWORD,
      email_confirm: true,
      user_metadata: { username: existing.username || DEFAULT_ADMIN_USERNAME, role: 'admin' },
    });
    if (error || !data.user) throw error || new Error('Unable to create Supabase Auth admin');
    await saveAdminCredentials({ username: existing.username || DEFAULT_ADMIN_USERNAME, authUserId: data.user.id, authEmail: email });
  } catch (error: any) {
    throw new Error('Supabase Auth is not ready. Run the updated sql/supabase_schema.sql in Supabase SQL Editor.', { cause: error });
  }
}

export async function createApp() {
  const app = express();

  // Do not block the HTTP server from starting if Supabase is temporarily unavailable.
  // Health checks can still pass, while authenticated routes will report the actual issue.
  try {
    await ensureAdminCredentialsTable();
  } catch (error) {
    console.error('Initial Supabase Auth setup failed; authenticated routes may be unavailable until Supabase is ready.', error);
  }
  // Lightweight security/performance middleware: avoid unnecessary framework dependencies.
  app.disable('x-powered-by');
  // The Tauri desktop shell is a local origin and therefore needs explicit CORS access
  // to the Railway API. Regular browser requests remain same-origin.
  app.use((req, res, next) => {
    const origin = String(req.headers.origin || '');
    if (origin === 'tauri://localhost' || origin === 'https://tauri.localhost' || origin === 'http://tauri.localhost') {
      res.setHeader('Access-Control-Allow-Origin', origin);
      res.setHeader('Access-Control-Allow-Credentials', 'true');
      res.setHeader('Access-Control-Allow-Headers', 'Authorization, Content-Type, X-IFC-Offline-Mutation, X-IFC-Request-ID, X-IFC-Device-ID');
      res.setHeader('Access-Control-Allow-Methods', 'GET,POST,PUT,DELETE,OPTIONS');
      res.setHeader('Vary', 'Origin');
      if (req.method === 'OPTIONS') return res.status(204).end();
    }
    next();
  });
  app.use((req, res, next) => {
    res.setHeader('X-Content-Type-Options', 'nosniff');
    res.setHeader('X-Frame-Options', 'DENY');
    res.setHeader('Referrer-Policy', 'strict-origin-when-cross-origin');
    res.setHeader('Permissions-Policy', 'camera=(), microphone=(), geolocation=()');
    if (process.env.NODE_ENV === 'production') {
      res.setHeader('Strict-Transport-Security', 'max-age=31536000; includeSubDomains');
    }
    if (req.path.startsWith('/api/')) {
      res.setHeader('Cache-Control', 'no-store');
    }
    }
  });

  // ----------------- V15 SYSTEM / DIAGNOSTICS -----------------
  app.get('/api/system/health', async (_req, res) => {
    const started = Date.now();
    try {
      const { data, error } = await supabaseServer.from('academy_settings').select('id,academy_name,updated_at').eq('id', 1).maybeSingle();
      if (error) throw error;
      res.json({
        status: 'ok',
        database: 'Supabase',
        latencyMs: Date.now() - started,
        academy: data?.academy_name || '',
        checkedAt: new Date().toISOString(),
      });
    } catch (error: any) {
      res.status(503).json({ status: 'degraded', database: 'unavailable', latencyMs: Date.now() - started, error: error?.message || 'Supabase unavailable', checkedAt: new Date().toISOString() });
    }
  });

  app.get('/api/system/audit', async (req, res) => {
    try {
      const limit = Math.min(500, Math.max(1, Number(req.query.limit || 100)));
      const { data, error } = await supabaseServer.from('audit_logs').select('*').order('created_at', { ascending: false }).limit(limit);
      if (error) throw error;
      res.json(data || []);
    } catch (error: any) {
      res.status(500).json({ error: error?.message || 'تعذر قراءة سجل العمليات.' });
    }
  });

  app.get('/api/system/trash', async (_req, res) => {
    try {
      const [players, coaches, payments, expenses, archives] = await Promise.all([
        supabaseServer.from('players').select('id,name,member_number,deleted_at').not('deleted_at', 'is', null).order('deleted_at', { ascending: false }).limit(500),
        supabaseServer.from('coaches').select('id,name,deleted_at').not('deleted_at', 'is', null).order('deleted_at', { ascending: false }).limit(500),
        supabaseServer.from('payments').select('id,invoice_number,player_name,amount,deleted_at').not('deleted_at', 'is', null).order('deleted_at', { ascending: false }).limit(500),
        supabaseServer.from('expenses').select('id,title,amount,deleted_at').not('deleted_at', 'is', null).order('deleted_at', { ascending: false }).limit(500),
        supabaseServer.from('monthly_archives').select('id,month_key,month_label,deleted_at').not('deleted_at', 'is', null).order('deleted_at', { ascending: false }).limit(500),
      ]);
      const err = [players, coaches, payments, expenses, archives].find(x => x.error)?.error;
      if (err) throw err;
      res.json({
        players: (players.data || []).map(x => ({ ...x, entityType: 'player' })),
        coaches: (coaches.data || []).map(x => ({ ...x, entityType: 'coach' })),
        payments: (payments.data || []).map(x => ({ ...x, entityType: 'payment' })),
        expenses: (expenses.data || []).map(x => ({ ...x, entityType: 'expense' })),
        archives: (archives.data || []).map(x => ({ ...x, entityType: 'archive' })),
      });
    } catch (error: any) {
      res.status(500).json({ error: error?.message || 'تعذر قراءة سلة المحذوفات.' });
    }
  });

  app.post('/api/system/trash/:entity/:id/restore', async (req, res) => {
    const tables: Record<string,string> = { player:'players', coach:'coaches', payment:'payments', expense:'expenses', archive:'monthly_archives' };
    const table = tables[String(req.params.entity)];
    if (!table) return res.status(400).json({ error: 'نوع العنصر غير مدعوم.' });
    try {
      const { error } = await supabaseServer.from(table).update({ deleted_at: null, updated_at: new Date().toISOString() }).eq('id', req.params.id);
      if (error) throw error;
      res.json({ success: true, entity: req.params.entity, id: req.params.id });
    } catch (error: any) {
      res.status(500).json({ error: error?.message || 'تعذر استرجاع العنصر.' });
    }
  });

  app.delete('/api/system/trash/:entity/:id/permanent', async (req, res) => {
    const tables: Record<string,string> = { player:'players', coach:'coaches', payment:'payments', expense:'expenses', archive:'monthly_archives' };
    const table = tables[String(req.params.entity)];
    if (!table) return res.status(400).json({ error: 'نوع العنصر غير مدعوم.' });
    try {
      const { error } = await supabaseServer.from(table).delete().eq('id', req.params.id);
      if (error) throw error;
      res.json({ success: true, permanent: true, entity: req.params.entity, id: req.params.id });
    } catch (error: any) {
      res.status(500).json({ error: error?.message || 'تعذر الحذف النهائي.' });
    }
  });

  app.get('/api/reports/range', async (req, res) => {
    try {
      const from = String(req.query.from || '').trim();
      const to = String(req.query.to || '').trim();
      if (!/^\d{4}-\d{2}-\d{2}$/.test(from) || !/^\d{4}-\d{2}-\d{2}$/.test(to)) return res.status(400).json({ error: 'استخدم التاريخ بصيغة YYYY-MM-DD.' });
      if (from > to) return res.status(400).json({ error: 'تاريخ البداية يجب أن يسبق تاريخ النهاية.' });
      const [p, e, s] = await Promise.all([
        supabaseServer.from('payments').select('*').gte('date', from).lte('date', to).is('deleted_at', null).order('date', { ascending: true }),
        supabaseServer.from('expenses').select('*').gte('date', from).lte('date', to).is('deleted_at', null).order('date', { ascending: true }),
        supabaseServer.from('player_sessions').select('*').gte('date', from).lte('date', to).order('date', { ascending: true }),
      ]);
      const err = [p,e,s].find(x => x.error)?.error;
      if (err) throw err;
      res.json({ from, to, payments: p.data || [], expenses: e.data || [], sessions: s.data || [],
        totals: { income: (p.data || []).reduce((n:any,x:any)=>n+Number(x.amount||0),0), expenses: (e.data || []).reduce((n:any,x:any)=>n+Number(x.amount||0),0), net: (p.data || []).reduce((n:any,x:any)=>n+Number(x.amount||0),0)-(e.data || []).reduce((n:any,x:any)=>n+Number(x.amount||0),0) }
      });
    } catch (error: any) {
      res.status(500).json({ error: error?.message || 'تعذر إنشاء التقرير.' });
    }
  });

  // ----------------- PLAYERS API -----------------
  app.get('/api/players', async (_req, res) => {
    try {
      const playersList = await getPlayers();
      res.json(playersList);
    } catch (error: any) {
      console.error('Failed to fetch players:', error);

    }
  });

  app.post('/api/players/bulk', async (req: AuthRequest, res) => {
    try {
      const players = Array.isArray(req.body?.players) ? req.body.players : [];

      const result = await bulkImportPlayers(players, String(req.body?.collectedBy || 'مسؤول الخزينة'), req.body?.registerSubscriptions !== false);
      let archive = null; try { archive = await refreshCurrentMonthArchive(String(req.body?.collectedBy || 'مسؤول الخزينة')); } catch (e) { console.warn('Archive refresh after player bulk skipped:', e); }
      res.json({ success: true, ...result, archive });
    } catch (error: any) {
      console.error('Failed bulk player import:', error);
      res.status(500).json({ error: error.message || 'تعذر استيراد اللاعبين بالجملة' });
    }
  });

  app.post('/api/players', async (req: AuthRequest, res) => {
    try {
      const newPlayer = await createPlayer(req.body);
      res.status(201).json(newPlayer);
    } catch (error: any) {
      console.error('Failed to create player:', error);
      res.status(500).json({ error: error.message || 'Failed to create player' });
    }

      await updatePlayer(req.params.id, req.body);
      const { data: player } = await supabaseServer.from('players').select('*').eq('id', req.params.id).maybeSingle();
      res.json({ success: true, id: req.params.id, player });
    } catch (error: any) {
      console.error('Failed to update player:', error);
      if (error instanceof SyncConflictError || error?.status === 409) { await recordSyncConflict(req, error); return res.status(409).json({ error: error.message, code: error.code || 'SYNC_CONFLICT', entityType: error.entityType, entityId: error.entityId, serverData: error.serverData || null }); }
      res.status(500).json({ error: error.message || 'Failed to update player' });
    }
  });

  app.delete('/api/players/:id', async (req: AuthRequest, res) => {
    try {
      await deletePlayer(req.params.id, req.body?._ifc_base_version);
      res.json({ success: true, id: req.params.id });
    } catch (error: any) {
      console.error('Failed to delete player:', error);
      if (error instanceof SyncConflictError || error?.status === 409) { await recordSyncConflict(req, error); return res.status(409).json({ error: error.message, code: error.code || 'SYNC_CONFLICT', entityType: error.entityType, entityId: error.entityId, serverData: error.serverData || null }); }
      res.status(500).json({ error: error.message || 'Failed to delete player' });
    }
  });

  app.put('/api/players/:playerId/sessions/:sessionId', async (req: AuthRequest, res) => {
    try {
      const { playerId, sessionId } = req.params;
      const { status, notes } = req.body;

      res.json({ success: true });
    } catch (error: any) {
      console.error('Failed to update session attendance:', error);
      res.status(500).json({ error: error.message || 'Failed to update attendance' });
    }

  // ----------------- PAYMENTS API -----------------
  app.get('/api/payments', async (_req, res) => {

      res.json(paymentsList);
    } catch (error: any) {

      res.status(500).json({ error: error.message || 'Failed to fetch payments' });

    } catch (error: any) {
      console.error('Failed to record payment:', error);
      res.status(500).json({ error: error.message || 'Failed to record payment' });

  });

  app.delete('/api/payments/:id', async (req: AuthRequest, res) => {
    try {
      await deletePayment(req.params.id);
      res.json({ success: true, id: req.params.id });
    } catch (error: any) {
      console.error('Failed to delete payment:', error);
      res.status(500).json({ error: error.message || 'Failed to delete payment' });
    }
  });

  // ----------------- EXPENSES API -----------------
  app.get('/api/expenses', async (_req, res) => {
    try {
      const expensesList = await getExpenses();
      res.json(expensesList);
    } catch (error: any) {

    try {
      const coaches = Array.isArray(req.body?.coaches) ? req.body.coaches : [];
      if (!coaches.length) return res.status(400).json({ error: 'لا توجد بيانات مدربين للاستيراد.' });
      if (coaches.length > 5000) return res.status(413).json({ error: 'ملف الاستيراد كبير جدًا. الحد الأقصى 5000 مدرب في العملية الواحدة.' });
      const result = await bulkImportCoaches(coaches);
      let archive = null; try { archive = await refreshCurrentMonthArchive('المدير العام (Admin)'); } catch (e) { console.warn('Archive refresh after coach bulk skipped:', e); }
      res.json({ success: true, ...result, archive });
    } catch (error: any) {
      console.error('Failed bulk coach import:', error);
      res.status(500).json({ error: error.message || 'تعذر استيراد المدربين بالجملة' });
    }
  });

  app.post('/api/coaches', async (req: AuthRequest, res) => {
    try {
      const newCoach = await createCoach(req.body);
      let archive = null; try { archive = await refreshCurrentMonthArchive('المدير العام (Admin)'); } catch (e) { console.warn('Archive refresh after coach create skipped:', e); }
      res.status(201).json({ ...newCoach, archive });
    } catch (error: any) {
      console.error('Failed to create coach:', error);
      res.status(500).json({ error: error.message || 'Failed to create coach' });
    }
  });

  app.put('/api/coaches/:id', async (req: AuthRequest, res) => {
    try {
      const coach = await updateCoach(req.params.id, req.body);
      res.json({ success: true, id: req.params.id, coach });
    } catch (error: any) {
      console.error('Failed to update coach:', error);
      if (error instanceof SyncConflictError || error?.status === 409) { await recordSyncConflict(req, error); return res.status(409).json({ error: error.message, code: error.code || 'SYNC_CONFLICT', entityType: error.entityType, entityId: error.entityId, serverData: error.serverData || null }); }
      res.status(500).json({ error: error.message || 'Failed to update coach' });
    }
  });

  app.delete('/api/coaches/:id', async (req: AuthRequest, res) => {
    try {
      await deleteCoach(req.params.id, req.body?._ifc_base_version);
      res.json({ success: true, id: req.params.id });
    } catch (error: any) {
      console.error('Failed to delete coach:', error);
      res.status(500).json({ error: error.message || 'Failed to delete coach' });
    }
  });

  // ----------------- ACADEMY RESET API -----------------
  // Destructive by design: removes academy operational data but keeps admin credentials and settings.
  app.post('/api/reset-academy-data', async (_req: AuthRequest, res) => {
    try {
      // The SQL function uses a transaction + advisory lock so a reset cannot race
      // with another device's sync/write and partially clear the academy.
      const result = await resetAcademyData();
      res.json(result);
    } catch (error: any) {
      console.error('Failed to reset academy data:', error);
      res.status(500).json({ error: error.message || 'تعذر تصفير بيانات الأكاديمية' });
    }
  });

  // ----------------- SETTINGS API -----------------
  app.get('/api/settings', async (_req, res) => {
    try {
      const settings = await getSettings();
      res.json(settings);
    } catch (error: any) {
      console.error('Failed to fetch settings:', error);
      res.status(500).json({ error: error.message || 'Failed to fetch settings' });
    }
  });

  app.put('/api/settings', async (req: AuthRequest, res) => {
    try {
      const updated = await updateSettings(req.body);
      res.json(updated);
    } catch (error: any) {
      console.error('Failed to update settings:', error);
      res.status(500).json({ error: error.message || 'Failed to update settings' });
    }
  });

  // ----------------- MONTHLY ARCHIVES API -----------------
  app.get('/api/archives', async (_req, res) => {
    try {
      try { await refreshCurrentMonthArchive('المدير العام (Admin)'); } catch (archiveRefreshError) { console.warn('Current archive refresh during read skipped:', archiveRefreshError); }
      const list = await getMonthlyArchives();
      res.json(list);
    } catch (error: any) {
      console.error('Failed to fetch monthly archives:', error);
      res.status(500).json({ error: error.message || 'Failed to fetch archives' });
    }
  });

  app.post('/api/archives', async (req: AuthRequest, res) => {
    try {
      const created = await createMonthlyArchive(req.body);
      res.status(201).json(created);
    } catch (error: any) {
      console.error('Failed to create monthly archive:', error);
      res.status(500).json({ error: error.message || 'Failed to create archive' });
    }
  });

  app.delete('/api/archives/:id', async (req: AuthRequest, res) => {
    try {
      await deleteMonthlyArchive(req.params.id);
      res.json({ success: true, id: req.params.id });
    } catch (error: any) {
      console.error('Failed to delete monthly archive:', error);
      res.status(500).json({ error: error.message || 'Failed to delete archive' });
    }
  });

  // ----------------- VITE MIDDLEWARE / STATIC FILES -----------------
  if (process.env.NODE_ENV !== 'production') {
    const isHmrDisabled = process.env.DISABLE_HMR === 'true';
    const vite = await createViteServer({
      server: {
        middlewareMode: true,
        hmr: isHmrDisabled ? false : undefined,
      },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    const distPath = path.join(process.cwd(), 'dist');
    app.use(express.static(distPath));
    app.get('*', (_req, res) => {
      res.sendFile(path.join(distPath, 'index.html'));
    });
  }

  return app;
}
